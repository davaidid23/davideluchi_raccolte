\poemtitle{Sguardo (A A.M.)}

\begin{verse}
    Sperso nel mio ingorgo intricato\\
    basta di sfuggita\\
    un tuo sorriso, delicato\\
    spiraglio\\
    per riavere entusiasmo di vita.\\!
    Così d'un tratto\\
    m'imbatto nel tuo sguardo randagio,\\
    taglio\\
    che apre il ventre\\
    mentre sono a mio agio.\\!
    E sento lucido, senza rime,\\
    l'intreccio d'anime del nostro abbraccio.
    
\end{verse}
