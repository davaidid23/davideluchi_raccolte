\poemtitle{Il prato}

\begin{verse}
    Da poco hanno tagliato l'erba.\\
    Non tutta, solo quella che permette\\
    agli anziani\\
    di camminare sereni\\
    con mani giunte sul fondoschiena.\\
    Senza inciampi.\\
    E intanto i ragazzi\\
    giocano nel bosco.\\!
    [[Uno è appena uscito dall'ombra verde\\
    ed è inciampato\\
    su un sasso\\
    nel prato.]]
\end{verse}
