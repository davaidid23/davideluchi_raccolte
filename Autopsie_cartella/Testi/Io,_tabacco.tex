\poemtitle{Tabacco}

\begin{verse}
    Tenere labbra\\
\end{verse}

\vspace{-0.9cm}
\begin{verse}
    aspirano \hfill a poco a poco
\end{verse}
\vspace{-0.9cm}

\begin{verse}
    l'anima\\
    d'una dolce sigaretta.\\!
    Ora consumata avidamente\\
    da fame meschina\\
    è gettata via,\\
    con leggerezza.\\!
    Così il mozzicone\\
    consumato\\
    arde la vita\\
    di cui si circonda:\\!
    eccolo, il carbone,\\
    che diviene la sua casa.
\end{verse}
