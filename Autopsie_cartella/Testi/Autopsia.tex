\poemtitle{Autopsia}

\begin{verse}
    Nella tetra quiete\\
    del crepuscolo\\
    volgo lo sguardo oltre l'orizzonte.\\!
    Lì, in quel buio\\
    intimo\\
    osservo allo specchio\\
    il mio corpo sventrato.\\!
    Così metto a fuoco\\
    \end{verse}
