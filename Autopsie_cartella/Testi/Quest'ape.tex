\poemtitle{Quest'ape}

\begin{verse}
    Quest'ape che danza e volteggia,\\
    (si sbagliano i pugili)\\
    e cerca l'aroma\\
    il colore\\
    il cuore del fiore,\\!
    come si muove elegante\\
    sull'uscio di casa    
\end{verse}
\vspace{-0.9cm}
\begin{verse}
    \hfill tesa, rientrante
\end{verse}
\vspace{-0.9cm}
\begin{verse}
    \hfill alle soglie del ring.
\end{verse}
