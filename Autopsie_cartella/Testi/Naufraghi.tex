\poemtitle{Naufraghi}

\begin{verse}
    Incespico tra relitti\\
    giunti alla mia riva\\
    lucenti, corrosi.\\
    Solo tepore di sole e loro zitti,\\
    mentre ne scruto i cocci\\
    raminghi.\\
    Mosaici,\\
    erosi dal tempo\\
    d'una vita subita,\\
    sbiaditi naufraghi!\\!
    V'accolgo e muoio\\
    io\\
    amante del mare.
\end{verse}
