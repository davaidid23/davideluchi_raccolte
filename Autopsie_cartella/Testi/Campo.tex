\poemtitle{Campo}

\begin{verse}
    Siena,\\
    aperto gioiello,\\
    spiraglio d'anima antica.\\
    Non si coglie di te\\
    che una carezza\\
    di sguardo,\\
    un sussurro inadeguato\\
    si respira,\\
    che di te racconta la fierezza intrisa\\
    nella storia d'ogni mattone.
\end{verse}
