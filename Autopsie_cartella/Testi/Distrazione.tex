\poemtitle{Distrazione}

\begin{verse}
    Quest orizzonte a strati\\
    di betulle fogliose e ingombranti,\\
    di montagne e dietro\\
    altre montagne.\\!
    Io s'una panchina\\
    all'ombra\\
    grazie a un Dio lungimirante\\
\end{verse}
