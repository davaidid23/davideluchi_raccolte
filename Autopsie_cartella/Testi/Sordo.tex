\poemtitle{Sordo}

\begin{verse}
    Volteggia leggiadra la vita\\
    tra centomila colori d'umore;\\
    ingenua s'insinua\\
    in stretti spiragli di luce,\\
    adagio, al tepore.\\!
    Di tanto in tanto\\
    l'olio del tempo\\
    lenisce e condanna,\\
    scalpellando nell'eterno\\
    gli intricati abissi\\
    del nostro labirinto.\\!
    E poi,\\
    Sordo schiocco di dita.
    
\end{verse}
