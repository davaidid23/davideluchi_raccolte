\poemtitle{Destino}

\begin{verse}
    Ho come l'impressione\\
    che questo oceano voglia evadere\\
    dalla sua simbiotica prigione\\
    e ogni onda\\
    sia un disperato grido d'aiuto\\
    che non oltrepassa
    le sue sabbiose sbarre.\\
    \end{verse}
    % \vspace{-0.9cm}
    \begin{verse}
    Noi guardiamo \hfill con gusto
    \end{verse}
    \vspace{-0.9cm}
    
    \begin{verse}
    e ne facciamo beffe\\
    bevendo per errore\\
    quelle salate lacrime.\\!
    Vilmente amiamo sguazzare\\
    nell'altrui sofferenza.
\end{verse}
