\poemtitle{Tressette}

\begin{verse}
    Sento gli schiamazzi\\
    da quella casa, lì\\
    a sinistra\\
    di poveri pazzi,\\
    che prima m'han salutato.\\
    Staranno giocando a tressette\\
    o forse è uno solo agitato\\
    che espelle fantasmi e timori.\\!
    Ma ci sono solo io a sentirvi\\
    qui fuori.
\end{verse}
