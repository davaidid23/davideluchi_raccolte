\poemtitle{Social}

\begin{verse}
    Ricordo d'aver abbracciato un albero\\
    troppo grosso per unire le dita.\\
    Mi chiedo se\\
    così concentrato e ostinato a vivere\\
    da tessere telai nella terra\\
    e così sociale\\
\end{verse}
\vspace{-0.9cm}
    \begin{verse}da essere un tutt'uno con il bosco \hfill ma solo\end{verse}
\vspace{-0.9cm}
\begin{verse}
    percepisca al tramonto\\
    di questa bestia a terra che l'accarezza\\
    di questo ammiratore con anima di corteccia\\
    proiettata al cielo.

\end{verse}
