\poemtitle{Che ne sarà}

\begin{verse}
Del sapore delle tue labbra,\\
Del profumo della tua pelle,\\
Della stretta del tuo abbraccio,\\
Dell'unione dei nostri corpi?\\!
Strana sensazione, la distanza.\\!
Carburante di Attesa e Fantasia,\\
Aspettative e Angosce,\\
dove tutto è in pausa,\\
in attesa della Vita.

\end{verse}
