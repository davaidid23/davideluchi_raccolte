\poemtitle{Naufrago (A D.L.)}

\begin{verse}
    Un sorriso che cela\\
    le mille sofferenze\\
    d'una vita\\
    a cui sei sempre stato grato.\\!
    Ti ringrazio, fratello,\\
    abile naufrago\\
    intento a contemplare le stelle.\\!
    Con te ho imparato\\
    i sapori della vita;\\
    che tu, ora, possa\\
    trovare quella di casa.\\!
    Seguila:\\
    ti porterà alla terra\\
    da cui sei stato strappato.   
    
\end{verse}
