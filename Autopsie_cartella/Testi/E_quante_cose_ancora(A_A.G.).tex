\poemtitle{E quante cose ancora... (A A.G.)}

\begin{verse}
    E quante cose ancora\\
    abbiamo da dirci.\\
    \end{verse}
    \vspace{-0.9cm}
    \begin{verse}
        Complici, osservando \hfill da lontano
    \end{verse}

    \vspace{-0.9cm}
\begin{verse}
    chi guarda il mondo\\
    con i soli occhi\\
    credendo di metterlo a fuoco.
\end{verse}
