\poemtitle{Canuta (A G.L.)}

\begin{verse}
    Patina d'apatia,\\
    unica medicina\\
    che lenisce il male del tempo,\\
    malattia.\\!
    Pupille offuscate da nebbia\\
    che separa mondo e ricordi\\
    in cui ignara ti rifugi.\\!
    Occhi assenti,\\
    e ricordo anche per te.
\end{verse}
