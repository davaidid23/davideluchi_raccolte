\poemtitle{Albero della vita}

\begin{verse}
    Un atavico dolore\\
\end{verse}
\vspace{-0.9cm}
    \begin{verse} che ci unisce \hfill tutti\end{verse}
\vspace{-0.9cm}
    \begin{verse} e battezza \hfill insieme\end{verse}
\vspace{-0.9cm}
\begin{verse}
    con avide radici.\\!
    Di questo si nutre\\
    il tetro albero della vita.
\end{verse}
