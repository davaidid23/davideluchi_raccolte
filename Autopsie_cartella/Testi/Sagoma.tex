\poemtitle{Sagoma}

\begin{verse}

    Un caldo affetto\\
    \end{verse}
    \vspace{-0.9cm}
    \begin{verse}che assume \hfill sbiadito \end{verse}
    \vspace{-0.9cm}
    \begin{verse}
    freddi contorni.\\
    Quel tepore\\
    di corpo\\
    che diventa gelida ombra\\
    e s'adagia\\
    sul fondo del cassetto\\
    delle mie ossessioni.


    % Il caldo affetto\\
    % assume sbiadito\\
    % freddi contorni.\\
    % E quel tepore\\
    % del tuo corpo\\
    % diventa fredda ombra\\
    % che s'adagia\\
    % sul fondo del cassetto\\
    % delle mie ossessioni.
\end{verse}
