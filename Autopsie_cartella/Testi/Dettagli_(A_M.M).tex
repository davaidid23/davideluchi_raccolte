\poemtitle{Dettagli (A M.M.)}

\begin{verse}
Il tuo sguardo sul mondo:\\
una spontanea dichiarazione di primavera,\\
dove instancabili pupille\\
scovano il fiore nell'asfalto.\\!
E in questo deserto,\\
urli al mondo\\
con ogni tuo premuroso gesto;\\
un minuzioso dettaglio d'amore\\
verso la vita,\\
verso l'amore.
\end{verse}
