\poemtitle{Clandestino}

\begin{verse}
Sempre\\
porto con me\\
il peso della vita.\\!
\end{verse}

\begin{verse}Né carezza di vento, \hfill cicatrice \end{verse}
\vspace{-0.9cm}
\begin{verse}né lacrima di pioggia, \hfill tenero livido \end{verse}
\vspace{-0.9cm}

\begin{verse}
mi distolgono\\
dalla nitida idea\\
che fui gettato\\
clandestino\\
nel mondo.\\!
\end{verse}
