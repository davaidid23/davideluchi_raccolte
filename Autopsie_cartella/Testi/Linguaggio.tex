\poemtitle{Linguaggio}

\begin{verse}
    Di tutte le vie le curve i marciapiedi\\
    ricordo dei visi i solchi\\
    i cenni dei vecchi\\
    dei bimbi gli occhi\\
    gli strani alfabeti,\\
    mattoni del muro del linguaggio.\\

    Eppure ho visto persone\\
    ridere e piangere\\
    nella mia stessa lingua.
\end{verse}
