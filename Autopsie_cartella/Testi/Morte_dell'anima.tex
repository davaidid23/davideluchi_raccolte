\poemtitle{Morte dell'anima}

\begin{verse}
    Sentimento mortale\\
    pervade fresche carni\\
    e le imputridisce.\\
    Figlia d'un dolor fatale\\
    mangia e logora corpi scarni,\\
    senza pietà agisce\\
    e risucchia la vita\\
    come una vecchia puttana.\\!
    È l'apatia il peggiore dei mali:\\
    genera uomini\\
    uccidendo umani.
\end{verse}
