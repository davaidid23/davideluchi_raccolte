\poemtitle{Ispirazione}

\begin{verse}
Nasce\\
nel dolore\\
della notte,\\
prepotente\\
mi dilania\\
per uscire.\\!
Il dito a sfera\\
incide\\
parole di morte\\
s'un bianco cadavere,\\
cedendogli spicchi d'anima.\\!
Così mi prosciugo,\\
scivolando nelle fredde\\
braccia della\\
solitudine.
\end{verse}
