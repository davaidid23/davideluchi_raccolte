\poemtitle{Frammento}

\begin{verse}
    Spesso mi porto appresso\\
    un frammento del trauma in corso\\
    e sono oppresso da questo peso\\
    e non riesco\\
    a calarmi nel contesto che vesto\\
    ma con un gesto son costretto\\
    a reclamare la mia appartenenza al mondo\\!
    pur non vivendo.    
\end{verse}
