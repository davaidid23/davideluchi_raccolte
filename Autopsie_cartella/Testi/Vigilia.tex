\poemtitle{Vigilia}

\begin{verse}
Vigilia d'un viaggio,\\
spiraglio di vita\\
passata tra paure e catene.\\!
È sognare, osservare,\\
svegliarsi e avere fede.\\
È l'ansia, fedele compagna,\\
\end{verse}

\vspace{-0.9cm}
\begin{verse}è paura materna, \hfill Eredità \end{verse}

\begin{verse}

\end{verse}
