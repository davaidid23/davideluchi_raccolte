\poemtitle{Tosse}

\begin{verse}
    Si rinnova nel mio viaggio\\
    la cagionevole speranza di scovare\\
    sotto macerie morali di vita,\\
    qualcuno che riesca a scrutare\\
    il mio abisso,\\
    senza voltarsi impaurita:\\
    sguardo fisso.\\!
    Questo artritico ingranaggio\\
    ricomincia la sua tosse.
\end{verse}
