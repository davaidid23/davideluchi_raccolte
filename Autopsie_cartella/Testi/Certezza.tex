\poemtitle{Certezza}

\begin{verse}
Ho la certezza\\
di non essere mai stato amato\\
se non per brevi sguardi\\
qualche carezza\\
e un paio di sorrisi.\\!
Se non per qualche dito sfiorato con delicatezza.\\!
Che sia la mia condanna?\\
Non capisco\\
questo tormento\\
e non lo accetto: mi ribello!\\!
\end{verse}

\begin{verse}Ancora voglio \hfill ne ho bisogno \end{verse}
