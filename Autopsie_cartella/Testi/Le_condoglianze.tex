\poemtitle{Le condoglianze}

\begin{verse}
    Odio le condoglianze.\\
    Un filtro verbale\\
    a un concetto banale\\
    che intercede con l'affondare\\
    le mani nel dolore.\\!
    Non riflette un'emozione\\
    ma un dovere:\\
    tradizione.\\!
    Quanto è più umano\\
    semplice autentico\\
    un banale e profondo\\
    \textit{mi dispiace}.
\end{verse}
