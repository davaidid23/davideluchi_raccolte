\poemtitle{Campana di vetro}

\begin{verse}
Ti osservo dalla mia campana di vetro.\\!
\end{verse}

\begin{verse}

\begin{verse}
    È in quegli occhi impauriti\\
    che accarezzo con delicatezza\\
    la tua intimità.\\!
    E te lo urlerei,\\
    ma l'aria è finita.
\end{verse}
