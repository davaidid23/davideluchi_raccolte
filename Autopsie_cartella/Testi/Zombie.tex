\poemtitle{Zombie (A M.C.)}

\begin{verse}
    Nell'anfratto più tetro, freddo,\\
    nell'angolo più umido, viscido, stagnante\\
    della mia mente\\
    io ti relego.\\!
    E nel buio del vuoto che hai dentro,\\
    ti lascio putrefatta\\
    a rimuginare\\
    sulla tua inutile miseria\\
    che seppellisci sotto sporchi\\
    teli di miserabile vergogna.\\!
    Marci fiera, a capo alto,\\
    per non inalare\\
    il puzzo acre del tuo marcire.
\end{verse}
